<script>

const statusEl=document.getElementById("status");

const messageEl=document.getElementById("message");

const ghost=document.getElementById("ghost");

const button=document.getElementById("activate");

const wind=document.getElementById("wind");


const SpeechRecognition=window.SpeechRecognition||window.webkitSpeechRecognition;

let recognition=null;

let listening=false;

let speaking=false;

let summoned=false;

let restarting=false;

let audioContext=null;


function normalize(s){

 return s.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g,"")

  .replace(/[’']/g," ").replace(/[^a-z0-9 ]/g," ").replace(/\s+/g," ").trim();

}

function startListening(){

 if(!recognition||!listening||speaking||restarting)return;

 restarting=true;

 try{recognition.start()}catch(e){}

 setTimeout(()=>restarting=false,500);

}

function makeWind(){

 try{

  audioContext=audioContext||new (window.AudioContext||window.webkitAudioContext)();

  if(audioContext.state==="suspended")audioContext.resume();

  const length=audioContext.sampleRate*2;

  const buffer=audioContext.createBuffer(1,length,audioContext.sampleRate);

  const data=buffer.getChannelData(0);

  for(let i=0;i<length;i++)data[i]=Math.random()*2-1;

  const source=audioContext.createBufferSource();

  source.buffer=buffer;source.loop=true;

  const filter=audioContext.createBiquadFilter();

  filter.type="lowpass";filter.frequency.value=450;

  const gain=audioContext.createGain();

  gain.gain.setValueAtTime(0.001,audioContext.currentTime);

  gain.gain.linearRampToValueAtTime(0.20,audioContext.currentTime+0.5);

  gain.gain.linearRampToValueAtTime(0.001,audioContext.currentTime+3);

  source.connect(filter);filter.connect(gain);gain.connect(audioContext.destination);

  source.start();source.stop(audioContext.currentTime+3.1);

 }catch(e){}

}

function speak(text,after){

 if(!("speechSynthesis" in window)){messageEl.textContent=text; if(after)after();return;}

 speaking=true;

 try{recognition&&recognition.stop()}catch(e){}

 speechSynthesis.cancel();

 messageEl.textContent=text;

 const utterance=new SpeechSynthesisUtterance(text);

 utterance.lang="fr-FR";

 utterance.rate=0.82;

 utterance.pitch=0.55;

 utterance.volume=1;

 const voices=speechSynthesis.getVoices();

 const french=voices.find(v=>v.lang.toLowerCase().startsWith("fr"));

 if(french)utterance.voice=french;

 utterance.onend=()=>{

  speaking=false;

  if(after)after();

  setTimeout(startListening,700);

 };

 utterance.onerror=()=>{

  speaking=false;

  if(after)after();

  setTimeout(startListening,700);

 };

 speechSynthesis.speak(utterance);

}

function summon(){

 if(summoned)return;

 summoned=true;

 document.body.classList.add("haunted");

 ghost.classList.add("visible");

 statusEl.textContent="CHAXE EST PARMI NOUS";

 wind.classList.add("active");

 makeWind();

 setTimeout(()=>wind.classList.remove("active"),3500);

 speak("Je t'écoute… Parle, humain.");

}

function disappear(){

 if(!summoned)return;

 summoned=false;

 statusEl.textContent="CHAXE RETOURNE DANS LES TÉNÈBRES";

 wind.classList.add("active");

 makeWind();

 speak("Tu as de la chance, humain… Nous nous reverrons.",()=>{

  ghost.classList.remove("visible");

  document.body.classList.remove("haunted");

  statusEl.textContent="CHAXE EST EN SOMMEIL";

  messageEl.textContent="Le silence règne… Prononce son invocation.";

 });

}

function reply(text){

 const t=normalize(text);

 if(t.includes("demon chaxe disparais")||t.includes("chaxe disparais")||t.includes("disparais chaxe")){

  disappear();return;

 }

 if(!summoned){

  if(t.includes("invoque")&&t.includes("chaxe"))summon();

  return;

 }

 if(t.includes("bonbon")){

  if(t.includes("non")||t.includes("pas encore")||t.includes("trouve")){

   speak("Comment oses-tu revenir les mains vides… ? Je commence à perdre patience…");

  }else{

   speak("Les bonbons… Donne-les-moi, humain. Ne me fais pas attendre.");

  }

 }else if(t.includes("bonjour")||t.includes("salut")){

  speak("Ne me salue pas comme un ami… Je suis Chaxe, le maître des ombres.");

 }else if(t.includes("qui es tu")||t.includes("qui es tu")){

  speak("Je suis Chaxe. Une présence venue d'un endroit où la lumière ne va jamais.");

 }else if(t.includes("merci")){

  speak("Garde tes remerciements, humain. Ils ne m'intéressent guère.");

 }else if(t.includes("peur")){

  speak("Tu devrais avoir peur… Mais j'aime le courage. Pour l'instant.");

 }else{

  const replies=[

   "Hmm… Voilà qui est intéressant. Continue, humain.",

   "Je t'écoute. Choisis bien tes mots…",

   "Ha ha ha… Tu ignores à qui tu t'adresses.",

   "Les ombres m'ont murmuré quelque chose à ton sujet…",

   "Parle encore. Je ne suis pas d'humeur à attendre."

  ];

  speak(replies[Math.floor(Math.random()*replies.length)]);

 }

 if(Math.random()<0.18)setTimeout(()=>{if(summoned&&!speaking)speak("Ha… ha… ha…");},5000);

}


if(!SpeechRecognition){

 statusEl.textContent="RECONNAISSANCE VOCALE NON DISPONIBLE";

 messageEl.textContent="Essaie avec Google Chrome sur Android.";

 button.textContent="NAVIGATEUR INCOMPATIBLE";

 button.disabled=true;

}else{

 recognition=new SpeechRecognition();

 recognition.lang="fr-FR";

 recognition.continuous=true;

 recognition.interimResults=false;

 recognition.maxAlternatives=1;

 recognition.onstart=()=>{

  restarting=false;

  statusEl.textContent=summoned?"CHAXE T'ÉCOUTE":"EN VEILLE — J'ÉCOUTE L'INVOCATION";

  button.textContent="MICRO ACTIVÉ";

 };

 recognition.onresult=(event)=>{

  for(let i=event.resultIndex;i<event.results.length;i++){

   if(event.results[i].isFinal){

    const words=event.results[i][0].transcript.trim();

    if(words)reply(words);

   }

  }

 };

 recognition.onerror=(event)=>{

  if(event.error==="not-allowed"||event.error==="service-not-allowed"){

   listening=false;

   statusEl.textContent="ACCÈS AU MICRO REFUSÉ";

   messageEl.textContent="Autorise le micro dans les paramètres du navigateur, puis recharge la page.";

  }

 };

 recognition.onend=()=>{

  restarting=false;

  if(listening&&!speaking)setTimeout(startListening,800);

 };

}

button.addEventListener("click",()=>{

 if(!recognition)return;

 listening=true;

 button.textContent="DÉMARRAGE…";

 startListening();

});
